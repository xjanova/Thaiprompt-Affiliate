=== วิธีที่ 1: รัน Command เดียว (แนะนำ) ===

cd /path/to/Thaiprompt-Affiliate && php artisan migrate --force


=== วิธีที่ 2: ใช้ Bash Script (อัตโนมัติ 100%) ===

cd /path/to/Thaiprompt-Affiliate
chmod +x URGENT_FIX_PRODUCTION.sh
./URGENT_FIX_PRODUCTION.sh


=== วิธีที่ 3: SQL โดยตรง (ถ้า artisan ไม่ทำงาน) ===

# Login MySQL
mysql -u root -p

# แล้วรัน:
USE your_database_name;

ALTER TABLE theme_settings ADD COLUMN footer_logo_animation ENUM('none', 'float', 'spin', 'bounce', 'pulse', 'swing') DEFAULT 'float' COMMENT 'Footer Logo Animation Style';

# ตรวจสอบ:
SHOW COLUMNS FROM theme_settings LIKE 'footer_logo_animation';


=== วิธีที่ 4: ใช้ SQL Script สำเร็จรูป ===

cd /path/to/Thaiprompt-Affiliate
mysql -u root -p < FIX_THEME_SETTINGS_COLUMNS.sql
